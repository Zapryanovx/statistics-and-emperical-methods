\chapter{Равномерна, условна и пълна вероятност}

\section{Равномерна (класическа) вероятност}

\begin{defbox}[~(Равномерна вероятност)]
За крайно вероятностно пространство с $n$ на брой равновъзможни
изхода, вероятността за всяко отделно събитие се дефинира като
\[
  \mathbb{P}(A) = \frac{\text{брой на благоприятните изходи за } A}{\text{общ брой възможни изходи}} = \frac{|A|}{n}.
\]
Основни характеристики: \emph{симетричност} (всички елементарни
изходи са еднакво вероятни), \emph{нормираност} (сумата от
вероятностите на елементарните изходи е~1) и \emph{простота}
(пресмятането на вероятности се свежда до броене на благоприятни
случаи).
\end{defbox}

\begin{rembox}[Формално представяне]
Нека $\Omega=\{s_1,\dots,s_n\}$ и $\mathbb{P}(s_i)=\tfrac1n$ за всяко
$i$. За произволно събитие $A\subseteq\Omega$: $\mathbb{P}(A)=
\tfrac{|A|}{n}$, където $|A|$ е броят на благоприятните изходи.
\end{rembox}

\begin{defbox}[~(Дискретна вероятност)]
Когато $\Omega=\{\omega_1,\omega_2,\dots\}$ е изброимо (но не
непременно с равновероятни изходи), дискретна вероятност се задава
чрез редица от числа $\{p_1,p_2,\dots\}$, $p_i\geq 0$,
$\sum_{i=1}^{\infty}p_i=1$, по правилото
\[
  \mathbb{P}(\{\omega_i\})=p_i,\qquad \mathbb{P}(A)=\sum_{\omega_i\in A}p_i,\quad A\subseteq\Omega.
\]
\end{defbox}

\begin{defbox}[~(Геометрична вероятност)]
Геометричната вероятност е пример за вероятностно разпределение
върху \emph{неизброимо} множество. За събитие $A$ в геометрично
пространство $\Omega$ (например $\Omega\subseteq\mathbb{R}^2$ с крайна
мярка $|\Omega|=\iint_\Omega dx<\infty$):
\[
  \mathbb{P}(A) = \frac{\text{мярка на благоприятната област } A}{\text{мярка на цялата област } \Omega} = \frac{|A|}{|\Omega|}.
\]
Основни характеристики: безкраен брой изходи, работа с непрекъснати
величини, използване на геометрични мерки (дължина за отсечки, площ
за фигури, обем за тела).
\end{defbox}

\begin{rembox}[Важно]
За геометрична вероятност, вероятността точков резултат да се случи
в дадена точка $x$ е винаги нула:
$\mathbb{P}(\{x\}) = \tfrac{|\{x\}|}{|\Omega|} = 0$,
защото площта на точка е нула. Вероятността зависи само от мярката
(площта/дължината/обема) на областта, а не от нейната локация или
форма. \emph{(Тази идея стои и в основата на алгоритмите на Монте
Карло за приблизително пресмятане на мерки/интеграли чрез случайно
семплиране.)}
\end{rembox}

\section{Условна вероятност}

\begin{defbox}[~(Условна вероятност)]
Нека $(\Omega,\mathcal{A},\mathbb{P})$ е вероятностно пространство и
$B\in\mathcal{A}$ е такова, че $\mathbb{P}(B)>0$. \textbf{Условната
вероятност} на събитие $A$ при условие $B$ се дефинира като
\[
  \mathbb{P}_B(A) = \mathbb{P}(A\mid B) = \frac{\mathbb{P}(A\cap B)}{\mathbb{P}(B)}.
\]
В частност, ако $A\subseteq B$, то $A\cap B=A$, откъдето
$\mathbb{P}(A\mid B) = \dfrac{\mathbb{P}(A)}{\mathbb{P}(B)}$ (коректно
между 0 и 1, защото $A\subseteq B\Rightarrow\mathbb P(A)\le\mathbb
P(B)$).
\end{defbox}

\begin{rembox}[Забележка]
Ако първоначалното вероятностно пространство е било
$(\Omega,\mathcal{A},\mathbb{P})$, то условната вероятност поражда
ново вероятностно пространство $(B,\mathcal{A}\cap B,\mathbb{P}(\cdot
\mid B))$, където новото пространство от елементарни събития е самото
$B$ (не цялото $\Omega$), $\mathcal A\cap B:=\{A\cap B:A\in\mathcal
A\}$ е следата на $\mathcal A$ върху $B$, а мярката е условната
вероятност $\mathbb P(\cdot\mid B)$ (не оригиналната $\mathbb P$).
\end{rembox}

\section{Независимост}

\begin{defbox}[~(Независимост на две събития)]
Две събития $A$ и $B$ се наричат \textbf{независими}, ако
\[
  \mathbb{P}(A\cap B) = \mathbb{P}(A)\cdot\mathbb{P}(B).
\]
\end{defbox}

\begin{corbox}[]
Ако $\mathbb{P}(A)>0$, то $A$ и $B$ са независими точно когато
$\mathbb{P}(B\mid A) = \mathbb{P}(B)$, т.е.\ независимостта означава,
че настъпването на $A$ не носи никаква информация за $B$.
\end{corbox}

\begin{defbox}[~(Взаимна независимост (в съвкупност))]
Дадени са събития $A_1,A_2,\dots,A_n$. Казваме, че те са
\textbf{независими взаимно (в съвкупност)}, ако за всяко непразно
$M\subseteq\{1,\dots,n\}$:
\[
  \mathbb{P}\Big(\bigcap_{i\in M}A_i\Big) = \prod_{i\in M}\mathbb{P}(A_i).
\]
Т.е.\ вероятността да се случи кое да е подмножество на $M$ от
събитията се разпада на произведението на вероятностите да се
случат съставните множества от $M$.
\end{defbox}

\begin{thmbox}[]{Теорема}
Нека $A_1,A_2,\dots,A_n$ са $n$ събития, такива че
$\mathbb{P}\big(\bigcap_{i=1}^{n}A_i\big)>0$. Тогава
\[
  \mathbb{P}\Big(\bigcap_{i=1}^{n}A_i\Big) = \mathbb{P}(A_n\mid \textstyle\bigcap_{i=1}^{n-1}A_i)\cdot\mathbb{P}(A_{n-1}\mid\textstyle\bigcap_{i=1}^{n-2}A_i)\cdots\mathbb{P}(A_2\mid A_1)\cdot\mathbb{P}(A_1).
\]
\end{thmbox}

\begin{proofbox}
По индукция по $n$.

\textbf{База.} За $n=1$: $\mathbb{P}(A_1)=\mathbb{P}(A_1)$ - тривиално.

\textbf{Индукционно предположение.} Допускаме, че равенството е
изпълнено за някое $n=k$.

\textbf{Индукционна стъпка.} Показваме го за $n=k+1$:
\[
  \mathbb{P}\Big(\bigcap_{i=1}^{k+1}A_i\Big) = \mathbb{P}\Big(A_{k+1}\cap\bigcap_{i=1}^{k}A_i\Big) = \mathbb{P}\Big(A_{k+1}\;\Big|\;\bigcap_{i=1}^{k}A_i\Big)\cdot\mathbb{P}\Big(\bigcap_{i=1}^{k}A_i\Big),
\]
и прилагаме индукционното предположение към последния множител. \qedbg
\end{proofbox}

\begin{corbox}[]
Ако $A_1,A_2,\dots,A_n$ са (взаимно) независими, то
\[
  \mathbb{P}\Big(\bigcap_{i=1}^{n}A_i\Big) = \prod_{i=1}^{n}\mathbb{P}(A_i).
\]
\end{corbox}

\section{Формула за пълната вероятност и формула на Бейс}

\begin{defbox}[~(Пълна група от събития)]
Нека $(\Omega,\mathcal{A},\mathbb{P})$ е вероятностно пространство.
Наборът от събития $\{A_1,A_2,\dots,A_n\}$ (или безкрайна изброима
редица $\{A_i\}_{i=1}^{\infty}$) се нарича \textbf{пълна група от
събития}, ако са изпълнени условията:
\begin{itemize}[nosep]
  \item \emph{несъвместимост (дизюнктност)}: $A_i\cap A_j=\varnothing$ за всички $i\neq j$ (никои две събития не могат да настъпят едновременно);
  \item \emph{изчерпателност}: $\bigcup_{i=1}^{n}A_i=\Omega$ (за краен случай) или $\bigcup_{i=1}^{\infty}A_i=\Omega$ (за безкраен изброим случай) - т.е.\ всички заедно покриват цялото вероятностно пространство.
\end{itemize}
\end{defbox}

\begin{thmbox}[~(Формула за пълната вероятност)]{Теорема}
Нека $(\Omega,\mathcal{A},\mathbb{P})$ е вероятностно пространство и
$H_1,\dots,H_n$ ($H_1,\dots,H_n,\dots$) е пълна група от събития.
Тогава за всяко събитие $A\in\mathcal{A}$ е в сила:
\[
  \mathbb{P}(A) = \sum_i \mathbb{P}(A\cap H_i) = \sum_i \mathbb{P}(A\mid H_i)\cdot\mathbb{P}(H_i),
\]
с конвенцията, че ако $\mathbb{P}(H_i)=0$, то приемаме $\mathbb{P}(A\mid H_i)=0$.
\end{thmbox}

\begin{proofbox}
Понеже $\{H_i\}$ е пълна група, $A = A\cap\Omega = A\cap
\bigcup_i H_i = \bigcup_i (A\cap H_i)$, като частите $A\cap H_i$ са
по двойки несъвместими събития (наследяват несъвместимостта на
$H_i$). Следователно, по $\sigma$-адитивност:
\[
  \mathbb{P}(A) = \mathbb{P}\Big(\bigcup_i (A\cap H_i)\Big) = \sum_i \mathbb{P}(A\cap H_i) = \sum_i \mathbb{P}(A\mid H_i)\,\mathbb{P}(H_i). \qedbg
\]
\end{proofbox}

\begin{thmbox}[~(Формула на Бейс)]{Теорема}
Нека $H_1,\dots,H_n$ ($H_1,\dots,H_n,\dots$) е пълна група от
събития в $\Omega$ и нека $A\subseteq\Omega$ е такова, че
$\mathbb{P}(A)>0$. Тогава за всяко $1\le k\le n$:
\[
  \mathbb{P}(H_k\mid A) = \frac{\mathbb{P}(A\mid H_k)\,\mathbb{P}(H_k)}{\mathbb{P}(A)} = \frac{\mathbb{P}(A\mid H_k)\,\mathbb{P}(H_k)}{\sum_i \mathbb{P}(A\mid H_i)\,\mathbb{P}(H_i)}.
\]
\end{thmbox}

\begin{proofbox}
От дефиницията на условна вероятност, приложена в двете посоки:
\[
  \mathbb{P}(A\cap H_k) = \mathbb{P}(A\mid H_k)\,\mathbb{P}(H_k)
  \qquad\text{и}\qquad
  \mathbb{P}(H_k\mid A)\,\mathbb{P}(A) = \mathbb{P}(H_k\cap A).
\]
Следователно
\[
  \mathbb{P}(H_k\mid A) = \frac{\mathbb{P}(H_k\cap A)}{\mathbb{P}(A)} = \frac{\mathbb{P}(H_k)\,\mathbb{P}(A\mid H_k)}{\mathbb{P}(A)},
\]
а по формулата за пълната вероятност заместваме $\mathbb{P}(A) =
\sum_j\mathbb{P}(H_j)\,\mathbb{P}(A\mid H_j)$ в знаменателя. \qedbg
\end{proofbox}
